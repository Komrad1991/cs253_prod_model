# ============================================================================
#  Демонстрация: циклы в базе правил и их разрешение.
#
#  Здесь намеренно собраны три вида циклов:
#    1) обратная дуга «потребность -> угроза» (CYC_3), замыкающая цепочку
#       угроза -> потребность -> угроза;
#    2) самопетля правила (CYC_SELF): заключение совпадает с посылкой;
#    3) обычная цепочка CYC_1 -> CYC_2 -> CYC_4 -> CYC_5, дающая ответ.
#
#  Что должно произойти:
#    * прямой вывод завершается (каждая подстановка правила срабатывает один
#      раз), повторные срабатывания блокируются учётом подстановок;
#    * обратный вывод упирается в цикл, отсекает ветвь (узел «цикл») и
#      продолжает поиск по альтернативному правилу — цель доказывается;
#    * без отсечения циклов обратный вывод зациклился бы на CYC_3.
#
#  Запуск:  pmcli run --kb data/demo_cycles.txt
# ============================================================================
kb "Демонстрация циклов"
description "Циклы: обратная дуга, самопетля; прямой и обратный вывод обязаны завершиться."

fact enemy(Haze) cf=0.95

rule CYC_1 cf=0.90
  if enemy(Haze)
  then threat(bullet_dps)
  why "Haze даёт пулевое давление."

rule CYC_2 cf=0.90
  if threat(bullet_dps)
  then need(anti_bullet)
  why "Пулевое давление требует защиты от пуль."

# Обратная дуга: потребность снова порождает ту же угрозу.
rule CYC_3 cf=0.99
  if need(anti_bullet)
  then threat(bullet_dps)
  why "НАМЕРЕННЫЙ ЦИКЛ: потребность подтверждает угрозу, которая её же породила."

# Самопетля: заключение совпадает с посылкой.
rule CYC_SELF cf=0.97
  if threat(bullet_dps)
  then threat(bullet_dps)
  why "НАМЕРЕННАЯ ПЕТЛЯ: правило выводит собственную посылку."

rule CYC_4 cf=0.95
  if need(anti_bullet)
  then pick(MetalSkin)
  why "Потребность закрывается Metal Skin."

rule CYC_5 cf=0.95
  if pick(MetalSkin)
  then answer(metal_skin)
  why "Итоговый ответ сценария."

goal answer(metal_skin)
goal threat(bullet_dps)
